{{- /* Builds the list of currently active concert announces, ready to render as
  • content: 1. dynamic announces from Directus (items/announces), already filtered by publish/expiry date; a cancelled linked event overrides the announce text with an auto-generated cancellation notice. Sorted by the event date; an announce without a linked event (e.g. one text for several concerts) sorts by its expiry_date, else publish_date — Directus' sort=event.date would put it last (NULLS LAST). 2. the legacy hand-written "announce" array in content/news.md, kept working as before. @returns {[]string} safe HTML strings, one per announce line. */}} {{- $items := slice }} {{- $dated := slice }} {{- $raw := partial "directus-announces.html" }} {{- range $raw.data }} {{- $sortKey := "" }} {{- with .event }}{{ $sortKey = .date }}{{ end }} {{- if not $sortKey }}{{ $sortKey = .expiry_date | default .publish_date | default "" }}{{ end }} {{- $text := .text }} {{- $link := .link }} {{- $line := "" }} {{- with .event }} {{- $d := time.AsTime .date }} {{- if eq (lower .status) "cancelled" }} {{- $line = printf "Концерт %s (%s) отменён." ($d | time.Format "02.01.2006") .town }} {{- else if $text }} {{- $line = $text | markdownify }} {{- with $link }}{{ $line = printf `%s` (htmlEscape .) $line | safeHTML }}{{ end }} {{- else }} {{- $prefix := "Концерт" }} {{- if eq .day_evening "дневной" }}{{ $prefix = "Дневной концерт" }} {{- else if eq .day_evening "вечерний" }}{{ $prefix = "Вечерний концерт" }} {{- end }} {{- $partA := printf "%s Михаила Щербакова" $prefix }} {{- if .with_mls }}{{ $partA = print $partA " и Михаила Стародубцева" }}{{ end }} {{- $partA = print $partA " " ($d | time.Format "02.01.2006") }} {{- with .time }}{{ $partA = print $partA " в " (printf "%.5s" .) }}{{ end }} {{- $partB := "" }} {{- with .town }}{{ $partB = . }}{{ end }} {{- with .place }}{{ $partB = print $partB ", " . }}{{ end }} {{- if .country }} {{- if .state }}{{ $partB = print $partB " (" .state ", " .country ")" }} {{- else }}{{ $partB = print $partB " (" .country ")" }} {{- end }} {{- end }} {{- $partB = print $partB "." }} {{- if $link }} {{- $line = printf `%s: %s` $partA (htmlEscape $link) $partB | safeHTML }} {{- else }} {{- $line = printf "%s: %s" $partA $partB }} {{- end }} {{- end }} {{- else }} {{- $line = $text | markdownify }} {{- with $link }}{{ $line = printf `%s` (htmlEscape .) $line | safeHTML }}{{ end }} {{- end }} {{- $dated = $dated | append (dict "key" (printf "%.10s" $sortKey) "line" $line) }} {{- end }} {{- range sort $dated "key" }}{{ $items = $items | append .line }}{{ end }} {{- with site.GetPage "news" }} {{- range .Params.announce }} {{- $items = $items | append (. | markdownify) }} {{- end }} {{- end }} {{- return $items }}
