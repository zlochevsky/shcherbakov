{{- /* Название песни — единое правило для каталогов, списков треков, страниц песен, концертов, «Словаря», циклов. Контекст — запись works (поля name, qualifier, fincipit). Возвращает dict: "html" — для видимого текста: выделение *так* в name → так (курсив), первая буква заглавная; "plain" — без разметки, с уточнением «: » (различает песни с одинаковым названием) — для , атрибутов data-*, поиска и сортировки. У песни без названия (name пусто) — fincipit. Есть ли у песни название — проверять по .name. */ -}} {{- $html := "" -}} {{- $plain := "" -}} {{- if .name -}} {{- $src := .name | strings.TrimSpace -}} {{- /* заглавной делаем первую букву, а не первый символ: «*завтра*», ««балтийские волны»» */ -}} {{- with findRESubmatch `^([^\pL\pN]*)(.)` $src 1 -}} {{- $pre := index . 0 1 -}} {{- $ch := index . 0 2 -}} {{- $src = print $pre (upper $ch) (strings.TrimPrefix (print $pre $ch) $src) -}} {{- end -}} {{- $html = replaceRE `\*([^*]+)\*` "<em>$1</em>" (htmlEscape $src) -}} {{- $plain = replaceRE `\*([^*]+)\*` "$1" $src -}} {{- with .qualifier }}{{ $plain = printf "%s: %s" $plain (strings.TrimSpace .) }}{{ end -}} {{- else -}} {{- $plain = .fincipit | default "" -}} {{- $html = htmlEscape $plain -}} {{- end -}} {{- return dict "html" ($html | safeHTML) "plain" $plain -}}
