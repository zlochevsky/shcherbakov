{{- /* Шапка страниц языка (главная языка, список переводов, страница перевода): верхнее табличное меню раздела и строка «хлебных крошек», отделяющая шапку от содержимого. Меню — params.nav главной страницы языка (content/<язык>/<язык>-index.md), на языке раздела, разметка как у partials/menu.html (nav.top-menu, бургер на мобильном); ссылки на текущую страницу нет; пункт с пустым href (страницы ещё нет) — неактивная ячейка. Крошки: русская главная › язык (самоназвание) › список переводов › перевод; последний пункт — текстом (aria-current), страницы, которых ещё нет, — тоже текстом. Контекст — dict "page" (текущая страница) "main" (главная страница языка) "crumb" (подпись последнего пункта — у страницы перевода и статьи; у главной и списка — пусто), "article" (true — статья раздела, не перевод: в крошках нет шага «список переводов»). */ -}} {{- $page := .page -}} {{- $main := .main -}} {{- $nav := $main.Params.nav | default slice -}} {{- $labels := $main.Params.labels | default dict -}} {{- $lang := $main.Params.lang -}} {{- /* адрес …/index.html Hugo записывает как …/ — сравниваем без index.html */ -}} {{- $here := strings.TrimSuffix "index.html" $page.RelPermalink -}} {{- $isHere := dict -}} {{- range $nav }}{{ with .href }}{{ $isHere = merge $isHere (dict . (eq (strings.TrimSuffix "index.html" (. | relURL)) $here)) }}{{ end }}{{ end -}} {{- partial "burger.html" }}
{{- range $nav }} {{- if not .href }}
{{ .label }}
{{- else if not (index $isHere .href) }}
{{ .label }}
{{- end }} {{- end }}
{{- /* крошки: русская главная — первый пункт nav; главная языка — второй; список переводов — третий */ -}} {{- $root := index $nav 0 -}} {{- $langHome := index $nav 1 -}} {{- $list := index $nav 2 -}} {{- $steps := slice (dict "label" $root.label "href" $root.href) (dict "label" ($main.Params.heading | default $lang) "short" ($labels.language | default "") "href" $langHome.href) (dict "label" ($labels.translations | default "") "href" $list.href) -}} {{- /* статья раздела (не перевод): крошки без шага «список переводов» */ -}} {{- if .article }}{{ $steps = first 2 $steps }}{{ end -}} {{- $last := 0 -}} {{- if index $isHere ($langHome.href | default "-") }}{{ $last = 1 }} {{- else if index $isHere ($list.href | default "-") }}{{ $last = 2 }} {{- else }}{{ $last = len $steps }}{{ end -}}
    {{- range $i, $s := $steps }} {{- if le $i $last }} {{- $label := cond (and (eq $i 1) $s.short) $s.short $s.label }} {{- if eq $i $last }}
  1. {{ $label }}
    2. {{- else if $s.href }}
  2. {{ $label }}
    3. {{- else }}
  3. {{ $label }}
    4. {{- end }} {{- end }} {{- end }} {{- if eq $last (len $steps) }}
  4. {{ $.crumb }}
    5. {{- end }}
